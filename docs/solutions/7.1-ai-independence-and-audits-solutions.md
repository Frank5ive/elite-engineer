```python
"""Invoice export helper, corrected. Python 3.12+ (itertools.batched is new in 3.12)."""
import hashlib
import hmac
import os
import re
import sqlite3
import subprocess
from decimal import Decimal, ROUND_HALF_UP
from itertools import batched

CENT = Decimal("0.01")
_SAFE_NAME = re.compile(r"[A-Za-z0-9_-]{1,64}")


def invoice_total(lines: list[dict], tax_rate: Decimal = Decimal("0.16")) -> Decimal:
    """lines: dicts with Decimal 'qty' and 'unit_price'."""
    subtotal = sum((l["qty"] * l["unit_price"] for l in lines), Decimal("0"))
    return (subtotal * (1 + tax_rate)).quantize(CENT, rounding=ROUND_HALF_UP)


def apply_discount(total: Decimal, pct: Decimal) -> Decimal:
    if not Decimal("0") <= pct <= Decimal("100"):
        raise ValueError("pct must be within [0, 100]")
    return (total * (1 - pct / 100)).quantize(CENT, rounding=ROUND_HALF_UP)


def find_invoices(db: sqlite3.Connection, customer: str) -> list[tuple]:
    return db.execute("SELECT id, total FROM invoices WHERE customer = ?", (customer,)).fetchall()


def add_tags(invoice_id: int, tag: str, tags: list | None = None) -> list:
    tags = [] if tags is None else tags
    tags.append((invoice_id, tag))
    return tags


def page(items: list, page_no: int, size: int = 10) -> list:
    if page_no < 0 or size < 1:
        raise ValueError("page_no >= 0 and size >= 1 required")
    start = page_no * size
    return items[start:start + size]


def verify_token(supplied: str, expected: str) -> bool:
    return hmac.compare_digest(supplied.encode(), expected.encode())


def hash_password(pw: str, *, n: int = 2**14) -> bytes:
    # n=2**14 keeps the demo fast. OWASP's scrypt minimum is n=2**17, r=8, p=1, which needs
    # maxmem=2**28 (the default limit is 32 MiB and hashlib raises ValueError at n=2**17).
    salt = os.urandom(16)  # store salt + parameters with the hash in real code
    return salt + hashlib.scrypt(pw.encode(), salt=salt, n=n, r=8, p=1)


def export_pdf(name: str) -> None:
    if not _SAFE_NAME.fullmatch(name):
        raise ValueError("invalid name")
    subprocess.run(["wkhtmltopdf", f"invoices/{name}.html", f"out/{name}.pdf"], check=True)


def chunks(rows, size: int = 100) -> list[tuple]:
    return list(batched(rows, size))
```

Alternative designs: use an ORM/query builder for SQL; use a vetted password library rather than hand-rolling scrypt parameters; return `(items, has_next)` for pagination by fetching `size + 1` internally.

## C. Challenge: AST scanner (reference sketch)

```python
import ast
from dataclasses import dataclass

@dataclass(frozen=True)
class Finding:
    line: int
    rule: str
    message: str

def _dotted(node):
    parts = []
    while isinstance(node, ast.Attribute):
        parts.append(node.attr); node = node.value
    if isinstance(node, ast.Name): parts.append(node.id)
    return ".".join(reversed(parts))

def scan(source: str) -> list[Finding]:
    out = []
    for n in ast.walk(ast.parse(source)):
        if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef)):
            for d in n.args.defaults + [x for x in n.args.kw_defaults if x]:
                if isinstance(d, (ast.List, ast.Dict, ast.Set)):
                    out.append(Finding(d.lineno, "mutable-default", "mutable default argument"))
        if isinstance(n, ast.Call):
            name = _dotted(n.func)
            if name in ("subprocess.run", "subprocess.Popen", "subprocess.call"):
                for kw in n.keywords:
                    if kw.arg == "shell" and isinstance(kw.value, ast.Constant) and kw.value.value is True:
                        out.append(Finding(n.lineno, "shell-true", "shell=True"))
            if name in ("hashlib.md5", "hashlib.sha1"):
                out.append(Finding(n.lineno, "weak-hash", name))
            if isinstance(n.func, ast.Attribute) and n.func.attr == "execute" and n.args:
                a = n.args[0]
                dyn = isinstance(a, ast.JoinedStr) or (isinstance(a, ast.BinOp) and isinstance(a.op, ast.Mod)) or \
                      (isinstance(a, ast.Call) and isinstance(a.func, ast.Attribute) and a.func.attr == "format")
                if dyn:
                    out.append(Finding(n.lineno, "sql-interpolation", "interpolated SQL"))
    return out
```

I ran this sketch against the lab listing: it flagged lines 40, 36, 59 and 55 and nothing for `execute('SELECT 1')` or `shell=False`. Known blind spots: data flow (query built on one line, executed on another), aliased imports (`from subprocess import run`), `shell=variable`. Trade-offs: the AST approach has far fewer false positives than regex but cannot follow values; real linters (`bandit`) cover more rules.
